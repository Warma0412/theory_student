\section{研究背景}

\begin{frame}{研究背景：经营看板的维度过载}
  \begin{block}{现实问题}
    GMV看板同时包含交易规模、价格成本、供给、支付、履约、口碑、地域和营销等大量指标。指标越多，越难判断哪些变量真正提供了独立信息。
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{常用做法}
      \begin{itemize}
        \item 业务经验筛选
        \item 边际相关与逐项检验
        \item 树模型重要性排序
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{主要不足}
      \begin{itemize}
        \item 高度相关代理重复入选
        \item 多重比较累积误报
        \item 重要性排序缺少错误比例控制
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{alertblock}{研究目标}
    在31项候选维度中筛选对下一期GMV具有增量预测信息的少数指标，并形成可调数量、可复核的监控集合。
  \end{alertblock}
\end{frame}

\begin{frame}{研究问题与预期贡献}
  \begin{enumerate}
    \item 如何从多表交易数据构造不重复计算GMV的卖家面板？
    \item 如何在混合分布、高度相关变量中控制目标错误发现率？
    \item 如何在确认集合内通过参数$K$控制最终展示数量？
    \item 月度结论在周度、卖家内变化、业务组和非线性模型下是否稳定？
    \item 与标准筛选方法相比，主流程的增量价值及各组件作用是什么？
  \end{enumerate}
  \vspace{0.3cm}
  \begin{block}{预期贡献}
    构建“受控确认—公平比较—参数化展示—多粒度复核”的统一框架，严格区分预测精度、选择稳定性、错误率控制与因果结论。
  \end{block}
\end{frame}

\begin{frame}{相关研究与方法依据}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{受控变量选择}
      \begin{itemize}
        \item BH错误发现率控制 [1]
        \item Knockoff过滤器 [2]
        \item Model-X Knockoff [3]
        \item 去随机化Knockoff [4]
        \item e-BH与e-value [5]
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{生成、预测与解释}
      \begin{itemize}
        \item MVR最小可重构性 [6]
        \item Deep Knockoffs [7]
        \item 深度核两样本检验 [8]
        \item XGBoost [9]
        \item TreeSHAP [10]
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{alertblock}{研究空白}
    现有方法较少把受控发现、统一预算基准、组件消融、展示数量和月周粒度复核纳入同一经营指标治理流程。
  \end{alertblock}
\end{frame}

\section{研究方法}

\begin{frame}{统一方法框架}
  \begin{center}
    \large
    原始多表数据
    $\rightarrow$ 自然粒度聚合
    $\rightarrow$ 卖家面板
    $\rightarrow$ Copula-MVR Knockoff
    $\rightarrow$ 60轮e-value
    $\rightarrow$ e-BH确认集合
    $\rightarrow$ 公平基准与参数$K$
  \end{center}
  \vspace{0.5cm}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{确认性主线}
      \begin{itemize}
        \item 随机化秩高斯Copula
        \item Ledoit-Wolf协方差
        \item MVR二阶Knockoff
        \item Lasso反对称统计量
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{补充验证}
      \begin{itemize}
        \item 时间外机器学习预测
        \item TreeSHAP与固定效应
        \item 卖家间/内与业务组
        \item 深度交换性诊断
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=8]{Copula-MVR：从混合边际到有效负对照}
  \begin{block}{秩高斯变换}
    \begin{equation*}
      Z_{ij}=\Phi^{-1}\left(\frac{r_{ij}-0.5}{n}\right)
    \end{equation*}
  \end{block}
  \begin{block}{二阶条件生成}
    \begin{equation*}
      \widetilde Z\mid Z\sim
      \mathcal N\left(
        Z-Z\widehat\Sigma^{-1}S,\;
        2S-S\widehat\Sigma^{-1}S
      \right)
    \end{equation*}
  \end{block}
  \begin{itemize}
    \item $\widehat\Sigma$：Ledoit-Wolf收缩协方差
    \item $S$：按MVR准则降低真变量与仿制变量可重构性
    \item 交换要求：$(X,\widetilde X)_{\mathrm{swap}(A)}\overset d=(X,\widetilde X)$
  \end{itemize}
  \begin{alertblock}{解释边界}
    $H_{0j}:Y\perp X_j\mid X_{-j}$；竞争统计量
    $W_j=|\widehat\beta_j|-|\widehat\beta_{j+p}|$。拒绝零假设只表示条件预测信息。
  \end{alertblock}
\end{frame}

\begin{frame}[shrink=10]{e-value聚合、e-BH与参数$K$}
  \begin{block}{第$m$轮变量$j$的e-value}
    \begin{equation*}
      e_j^{(m)}
      =p\frac{\mathbf 1\{W_j^{(m)}\ge T^{(m)}\}}
      {1+\sum_{k=1}^{p}\mathbf 1\{W_k^{(m)}\le-T^{(m)}\}}
    \end{equation*}
  \end{block}
  \begin{equation*}
    \bar e_j=\frac{1}{60}\sum_{m=1}^{60}e_j^{(m)},\qquad
    \bar e_{(k)}\ge\frac{p}{\alpha_{\mathrm{eBH}}k}
  \end{equation*}
  \begin{itemize}
    \item 单轮$\alpha_{\mathrm{kn}}=0.10$；最终$\alpha_{\mathrm{eBH}}=0.20$
    \item e-BH承担正式FDR结论，单轮频率只描述随机稳定性
    \item $K^*=\min\{K:C_K\ge90\%,\ \mathrm{RMSE}_K\le1.005\,\mathrm{RMSE}_{\min}\}$
  \end{itemize}
\end{frame}

\section{计划与产出}

\begin{frame}[shrink=6]{研究创新点与方法主线}
  \begin{enumerate}
    \item \textbf{单一确认主线：}Copula-MVR、反对称$W$、e-value与e-BH共同承担正式发现。
    \item \textbf{确认—展示分离：}参数$K$只压缩确认池，不反向改变FDR阈值。
    \item \textbf{分层证据：}月度确认，周度、业务组、卖家间/内与深度诊断负责复核。
    \item \textbf{明确否决：}未通过交换性诊断的深度生成器不得用于FDR声明。
    \item \textbf{公平评价：}统一窗口、预算、预测器与测试期，并用消融和模拟解释组件作用。
  \end{enumerate}
\end{frame}
